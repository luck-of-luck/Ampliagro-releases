  describe('os menus', () => {
    it('"criar no chat" abre com planilha e grafico', () => {
      window.IaRail.passar('criar-no-chat');
      const menu = document.querySelector('.ia-rail-menu');
      expect(menu).toBeTruthy();
      const rotulos = [...menu.querySelectorAll('button')].map((b) => b.textContent);
      expect(rotulos).toEqual(['Planilha', 'Gráfico']);
      /* E o botao diz que esta aberto — sem isto, o produtor não sabe se
         carregou no botão ou se a lista veio de outro sítio. */
      expect(rail.querySelector('[data-id="criar-no-chat"]').getAttribute('aria-expanded')).toBe(
        'true'
      );
    });

    it('"criar no painel" abre com as DUAS formas de montar', () => {
      window.IaRail.passar('criar-no-painel');
      const menu = document.querySelector('.ia-rail-menu');
      const rotulos = [...menu.querySelectorAll('button')].map((b) => b.textContent);
      expect(rotulos).toContain('Com os widgets que já existem');
      expect(rotulos).toContain('Widgets vazios + seus valores');
    });

    /* O menu vive IRMÃO do item: o `overflow:hidden` do item existe para o
       texto não vazar, e um menu dentro dele é cortado no mesmo quadro. */
    it('o menu é irmão do botão, e nunca filho', () => {
      window.IaRail.passar('criar-no-chat');
      const item = rail.querySelector('[data-id="criar-no-chat"]');
      const menu = document.querySelector('.ia-rail-menu');
      expect(item.contains(menu), 'o menu esta dentro do item e vai ser cortado').toBe(false);
      expect(menu.parentElement).toBe(rail);
    });

    it('abrir um menu fecha o anterior', () => {
      window.IaRail.passar('criar-no-chat');
      expect(document.querySelectorAll('.ia-rail-menu').length).toBe(1);
      window.IaRail.passar('criar-no-painel');
      expect(document.querySelectorAll('.ia-rail-menu').length).toBe(1);
    });

    it('a ferramenta escolhida é executada e o menu fecha', () => {
      window.IaRail.passar('criar-no-chat');
      clique(document.querySelectorAll('.ia-rail-menu button')[0]);
      expect(window.AmpliaChamado.pedirChat).toHaveBeenCalledTimes(1);
      expect(document.querySelector('.ia-rail-menu')).toBeNull();
    });

    /* Um `onclick` com um nome de função mal escrito é o defeito mais caro
     * desta barra: o botão fica activo, o clique não faz nada, e a única pista
     * é um `TypeError` na consola de quem sabe abrir a consola. */
    it('o onclick de cada ferramenta aponta para uma função que existe', () => {
      for (const menu of ['criar-no-chat', 'criar-no-painel']) {
        window.IaRail.passar(menu);
        for (const b of document.querySelectorAll('.ia-rail-menu button')) {
          const onclick = b.getAttribute('onclick');
          expect(onclick, `${menu}: botao sem onclick`).toBeTruthy();
          const nome = (onclick.match(/IaRail\.(\w+)\(/) || [])[1];
          expect(nome, `onclick mal formado: ${onclick}`).toBeTruthy();
          expect(typeof window.IaRail[nome], `IaRail.${nome} nao existe`).toBe('function');
        }
      }
    });
  }
